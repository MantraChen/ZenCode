// This valid program must not receive spelling or punctuation warnings.
public class CorrectCounterexamples {
    public static void main(String[] args) {
        System.out.println(gte(2, 1));
        String text = "中文；（）＝";
        char symbol = '；';
        // 中文注释；不会替换
        System.out.println(text + symbol);
    }
    public static boolean gte(int left, int right) { return left >= right; }
}
